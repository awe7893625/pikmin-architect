### 3. 在 `userContentController` 函數後添加以下函數（第 152 行後）
```swift
// 檢查並更新授權狀態
private func checkAndUpdateAuthStatus() {
    authManager.checkAuth { [weak self] authorized in
        guard let self = self else { return }
        DispatchQueue.main.async {
            self.updateTrialStatusInWebView()
        }
    }
}

// 激活授權碼
private func activateLicense(key: String) {
    authManager.activateLicense(key: key) { [weak self] success, message in
        guard let self = self else { return }
        DispatchQueue.main.async {
            if success {
                self.webView?.evaluateJavaScript("alert('✅ 授權激活成功！')")
            } else {
                self.webView?.evaluateJavaScript("alert('❌ \(message ?? "激活失敗")')")
            }
            self.updateTrialStatusInWebView()
        }
    }
}

// 更新 WebView 中的剩餘次數顯示
private func updateTrialStatusInWebView() {
    let remaining = authManager.remainingTrial
    let isPaid = authManager.isAuthorized && authManager.authStatus == .authorized
    let js = "updateTrialStatus(\(remaining), \(isPaid))"
    webView?.evaluateJavaScript(js)
}
```
