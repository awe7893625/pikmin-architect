#### 2. 修改 init() 方法（第 48-54 行）

```swift
override init() {
    super.init()
    // 自動載入預設軌跡
    DispatchQueue.main.asyncAfter(deadline: .now() + 1.0) {
        self.loadDefaultTrack()
    }
    // 啟動時檢查授權狀態
    DispatchQueue.main.asyncAfter(deadline: .now() + 2.0) {
        self.authManager.checkAuth { [weak self] _ in
            self?.updateTrialStatusInWebView()
        }
    }
    // 監聽授權狀態變化
    authManager.$remainingTrial
        .receive(on: DispatchQueue.main)
        .sink { [weak self] _ in
            self?.updateTrialStatusInWebView()
        }
        .store(in: &cancellables)
    
    authManager.$authStatus
        .receive(on: DispatchQueue.main)
        .sink { [weak self] _ in
            self?.updateTrialStatusInWebView()
        }
        .store(in: &cancellables)
}
```

#### 3. 修改 tp 和 startRoute（第 131-140 行）

將：
```swift
case "tp":
    // 停止所有移動模式
    self.stopTimerOnly()
    self.stopStuckCheck()
    self.isUsingRealTrack = false
    // 執行瞬移
    self.teleport(lat: dict["la"] as? Double ?? 0, lon: dict["lo"] as? Double ?? 0)
case "startRoute":
    let pts = dict["pts"] as? [[Double]] ?? []
    self.startCruise(points: pts, kmh: 18.0)
```

改為：
```swift
case "tp":
    // 檢查授權後執行瞬移
    self.authManager.performActionWithAuth { [weak self] success, errorMessage in
        guard let self = self else { return }
        if success {
            self.stopTimerOnly()
            self.stopStuckCheck()
            self.isUsingRealTrack = false
            self.teleport(lat: dict["la"] as? Double ?? 0, lon: dict["lo"] as? Double ?? 0)
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
                self.updateTrialStatusInWebView()
            }
        } else {
            DispatchQueue.main.async {
                let msg = (errorMessage ?? "試用次數已用完，請購買授權").replacingOccurrences(of: "'", with: "\\'")
                self.webView?.evaluateJavaScript("alert('\(msg)'); updateTrialStatus(0, false);")
            }
        }
    }
case "startRoute":
    // 檢查授權後執行散花
    self.authManager.performActionWithAuth { [weak self] success, errorMessage in
        guard let self = self else { return }
        if success {
            let pts = dict["pts"] as? [[Double]] ?? []
            self.startCruise(points: pts, kmh: 18.0)
            DispatchQueue.main.asyncAfter(deadline: .now() + 0.5) {
                self.updateTrialStatusInWebView()
            }
        } else {
            DispatchQueue.main.async {
                let msg = (errorMessage ?? "試用次數已用完，請購買授權").replacingOccurrences(of: "'", with: "\\'")
                self.webView?.evaluateJavaScript("alert('\(msg)'); updateTrialStatus(0, false);")
            }
        }
    }
```
